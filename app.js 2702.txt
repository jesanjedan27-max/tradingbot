// Deriv DigitDiff Bot — OAuth login + consecutive-pair strategy
document.addEventListener("DOMContentLoaded", () => {

  // ── Your registered Deriv app ─────────────────────────────────────────────
  const DERIV_APP_ID = "33wZZKTFZrmsZgFaAH53Z";
  const WS_ENDPOINT  = `wss://ws.derivws.com/websockets/v3?app_id=${DERIV_APP_ID}`;
  const OAUTH_URL    = `https://oauth.deriv.com/oauth2/authorize?app_id=${DERIV_APP_ID}&l=EN&brand=deriv`;
  const SYMBOL       = "R_100";
  const SWITCH_MULTIPLIERS = [1, 4.05 / 0.35, 52.63 / 0.35];

  // ── DOM refs ──────────────────────────────────────────────────────────────
  const $ = id => document.getElementById(id);

  const loginScreen         = $("loginScreen");
  const botScreen           = $("botScreen");
  const loginBtn            = $("loginBtn");
  const logoutBtn           = $("logoutBtn");
  const accountDisplay      = $("accountDisplay");
  const accountTypeEl       = $("accountType");
  const accountSelectorWrap = $("accountSelectorWrap");
  const accountSelector     = $("accountSelector");
  const modeIndicator       = $("modeIndicator");

  const startBtn    = $("start");
  const pauseBtn    = $("pause");
  const stopBtn     = $("stop");
  const resetBtn    = $("reset");
  const demoBtn     = $("demoBtn");
  const liveBtn     = $("liveBtn");

  const priceEl     = $("price");
  const balanceEl   = $("balance");
  const profitEl    = $("profit");
  const levelEl     = $("level");
  const lastDigitEl = $("lastDigit");
  const logEl       = $("log");
  const stakeInput  = $("stakeInput");

  // ── Auth state ────────────────────────────────────────────────────────────
  let accounts      = [];
  let activeToken   = null;
  let activeLoginid = null;

  // ── Bot state ─────────────────────────────────────────────────────────────
  let ws      = null;
  let running = false;
  let paused  = false;

  let lastPayoutRatio  = null;
  let recoveryLoss     = 0;
  let currentStake     = 0;
  let lastBalance      = null;
  let ladder           = 0;
  let totalProfit      = 0;

  let waitingProposal  = false;
  let proposalVariants = null;
  let proposalAttempt  = 0;
  let activeContractId = null;
  let probing          = false;
  let pendingBarrier   = null;

  let stakeMode     = "switch";
  let stakeLevelIdx = 0;

  let targetPair      = null;
  let phase           = "scan_ab";
  let seqA            = null;
  let seqB            = null;
  let settlementDigit = null;
  let captureNextTick = false;

  const LOG_MAX_ENTRIES  = 1200;
  const TICK_FLUSH_MS    = 60;
  const TICK_BATCH_LIMIT = 200;
  let tickBuffer     = [];
  let tickFlushTimer = null;

  // ═══════════════════════════════════════════════════════════════════════════
  // OAUTH — Deriv appends tokens to your redirect URL after login:
  // https://jesanjedan27-max.github.io/tradingbot/?token1=xxx&acct1=CR123&cur1=USD
  // ═══════════════════════════════════════════════════════════════════════════

  function parseOAuthCallback() {
    const params = new URLSearchParams(window.location.search);
    const parsed = [];
    let i = 1;
    while (params.get(`token${i}`)) {
      parsed.push({
        token:      params.get(`token${i}`),
        loginid:    params.get(`acct${i}`) || `Account ${i}`,
        currency:   params.get(`cur${i}`)  || "USD",
        is_virtual: (params.get(`acct${i}`) || "").startsWith("VRT")
      });
      i++;
    }
    return parsed;
  }

  function saveAccounts(accs) {
    localStorage.setItem("deriv_accounts", JSON.stringify(accs));
  }

  function loadAccounts() {
    try { return JSON.parse(localStorage.getItem("deriv_accounts") || "[]"); }
    catch { return []; }
  }

  function clearSession() {
    localStorage.removeItem("deriv_accounts");
    accounts = []; activeToken = null; activeLoginid = null;
    window.history.replaceState({}, document.title, window.location.pathname);
  }

  function buildAccountSelector(accs) {
    accountSelector.innerHTML = "";
    accs.forEach((acc, idx) => {
      const opt = document.createElement("option");
      opt.value = idx;
      opt.textContent = `${acc.loginid} — ${acc.is_virtual ? "Demo" : "Real"} (${acc.currency})`;
      accountSelector.appendChild(opt);
    });
  }

  function applySelectedAccount() {
    const idx = Number(accountSelector.value);
    const acc = accounts[idx];
    if (!acc) return;
    activeToken   = acc.token;
    activeLoginid = acc.loginid;

    accountDisplay.textContent = acc.loginid;
    accountTypeEl.textContent  = acc.is_virtual ? "Demo" : "Real";

    if (acc.is_virtual) {
      demoBtn.classList.add("active");
      liveBtn.classList.remove("active");
      modeIndicator.textContent = "JESAN 💲 MODE - DEMO";
      modeIndicator.className   = "mode-indicator demo";
    } else {
      liveBtn.classList.add("active");
      demoBtn.classList.remove("active");
      modeIndicator.textContent = "JESAN 💲 MODE - LIVE";
      modeIndicator.className   = "mode-indicator live";
    }
  }

  function showLoginScreen() {
    loginScreen.style.display = "";
    botScreen.style.display   = "none";
  }

  function showBotScreen() {
    loginScreen.style.display = "none";
    botScreen.style.display   = "";
  }

  function init() {
    // Check if Deriv just redirected back with tokens in the URL
    const fromOAuth = parseOAuthCallback();
    if (fromOAuth.length > 0) {
      accounts = fromOAuth;
      saveAccounts(accounts);
      window.history.replaceState({}, document.title, window.location.pathname);
    } else {
      accounts = loadAccounts();
    }

    if (accounts.length === 0) { showLoginScreen(); return; }

    showBotScreen();

    if (accounts.length > 1) {
      accountSelectorWrap.style.display = "";
      buildAccountSelector(accounts);
      accountSelector.addEventListener("change", () => {
        if (ws) { ws.close(); ws = null; }
        applySelectedAccount();
        appendLogLine("Account changed — press Start to reconnect.", "yellow");
      });
    }

    applySelectedAccount();
    createSwitchUI();
    appendLogLine("Logged in ✓ — press Start to begin trading.", "lime");
  }

  // ─────────────────────────────────────────────────────────────────────────
  // LOGGING
  // ─────────────────────────────────────────────────────────────────────────

  function appendLogLine(message, color = "#fff") {
    const entry = document.createElement("div");
    entry.style.color = color;
    entry.textContent = message;
    logEl.appendChild(entry);
    while (logEl.children.length > LOG_MAX_ENTRIES) logEl.removeChild(logEl.firstChild);
    logEl.scrollTop = logEl.scrollHeight;
  }

  function startTickFlush() {
    if (tickFlushTimer) return;
    tickFlushTimer = setInterval(() => {
      if (!tickBuffer.length) return;
      const fragment = document.createDocumentFragment();
      tickBuffer.splice(0, TICK_BATCH_LIMIT).forEach(({ price, digit }) => {
        const row = document.createElement("div");
        row.style.color = "#7dd3fc";
        row.textContent = `Tick ${Number(price).toFixed(2)} → ${digit ?? "-"}`;
        fragment.appendChild(row);
      });
      logEl.appendChild(fragment);
      while (logEl.children.length > LOG_MAX_ENTRIES) logEl.removeChild(logEl.firstChild);
      logEl.scrollTop = logEl.scrollHeight;
    }, TICK_FLUSH_MS);
  }

  function stopTickFlush() {
    if (!tickFlushTimer) return;
    clearInterval(tickFlushTimer);
    tickFlushTimer = null;
  }

  // ─────────────────────────────────────────────────────────────────────────
  // HELPERS
  // ─────────────────────────────────────────────────────────────────────────

  function digitFromPrice(price) {
    const v = Number(price);
    if (Number.isNaN(v)) return null;
    const s = v.toFixed(2);
    return Number(s[s.length - 1]);
  }

  function updateBalance(value) {
    if (typeof value !== "number" || Number.isNaN(value)) return;
    lastBalance = value;
    if (balanceEl) balanceEl.textContent = value.toFixed(2);
  }

  function parsedStakeLevels() {
    const base = Number(stakeInput?.value) || 0.35;
    return SWITCH_MULTIPLIERS.map(m => Number((base * m).toFixed(2)));
  }

  function currentSwitchStake() {
    const levels = parsedStakeLevels();
    return levels[Math.min(stakeLevelIdx, levels.length - 1)];
  }

  function baseStakeAmount() {
    return stakeMode === "switch" ? currentSwitchStake() : Number(stakeInput?.value || 0.35);
  }

  function recoveryStake(liveRatio) {
    if (stakeMode === "switch") return baseStakeAmount();
    const base = Number(stakeInput?.value || 0.35);
    if (recoveryLoss > 0 && liveRatio > 1.01)
      return Number(Math.max(base, recoveryLoss / (liveRatio - 1)).toFixed(2));
    return Number(base.toFixed(2));
  }

  // ── Switch-mode UI ────────────────────────────────────────────────────────

  function createSwitchUI() {
    if (!stakeInput || document.getElementById("stakeModeBtn")) return;
    const container = stakeInput.parentElement;
    if (!container) return;

    const modeBtn = document.createElement("button");
    modeBtn.id = "stakeModeBtn";
    modeBtn.style.cssText =
      "display:block;width:100%;margin-bottom:8px;padding:6px 10px;" +
      "border-radius:6px;border:none;cursor:pointer;font-weight:bold;" +
      "background:#7c3aed;color:#fff;font-size:13px;";

    const levelsRow = document.createElement("div");
    levelsRow.style.cssText = "margin-bottom:8px;";
    const levelsLbl = document.createElement("div");
    levelsLbl.style.cssText = "color:#94a3b8;font-size:11px;margin-bottom:4px;";
    levelsLbl.textContent = "Auto stake ladder (scales with base stake):";
    const levelsDisplay = document.createElement("div");
    levelsDisplay.style.cssText =
      "padding:5px 8px;border-radius:6px;border:1px solid #374151;" +
      "background:#111827;color:#38bdf8;font-size:12px;font-weight:bold;letter-spacing:0.5px;";

    const refreshLevels = () => {
      levelsDisplay.textContent = parsedStakeLevels().map((v, i) => `Lvl ${i+1}: $${v}`).join("  →  ");
    };

    levelsRow.append(levelsLbl, levelsDisplay);
    stakeInput.addEventListener("input", () => { if (stakeMode === "switch") refreshLevels(); });

    const refreshModeUI = () => {
      if (stakeMode === "switch") {
        modeBtn.textContent = "⚡ MODE: STAKE SWITCH";
        modeBtn.style.background = "#7c3aed";
        levelsRow.style.display = "";
        refreshLevels();
      } else {
        modeBtn.textContent = "📈 MODE: MARTINGALE";
        modeBtn.style.background = "#059669";
        levelsRow.style.display = "none";
      }
    };

    modeBtn.onclick = () => {
      stakeMode = stakeMode === "switch" ? "martingale" : "switch";
      stakeLevelIdx = 0; recoveryLoss = 0;
      refreshModeUI();
      appendLogLine(`Stake mode → ${stakeMode === "switch" ? "STAKE SWITCH" : "MARTINGALE"}`, "#a78bfa");
    };

    container.insertAdjacentElement("beforebegin", levelsRow);
    container.insertAdjacentElement("beforebegin", modeBtn);
    refreshModeUI();
  }

  // ─────────────────────────────────────────────────────────────────────────
  // STRATEGY
  // ─────────────────────────────────────────────────────────────────────────

  function nextPairFromResultDigit(digit) {
    if (digit === 9) return [0, 1];
    if (digit >= 0 && digit <= 8) return [digit, digit + 1];
    return null;
  }

  function resetSequence(nextTarget = null) {
    phase = "scan_ab"; seqA = null; seqB = null;
    settlementDigit = null; captureNextTick = false;
    waitingProposal = false; proposalVariants = null; proposalAttempt = 0;
    activeContractId = null; probing = false; pendingBarrier = null;
    targetPair = nextTarget;
    appendLogLine(
      nextTarget
        ? `Next scan: looking for pair [${nextTarget[0]},${nextTarget[1]}]`
        : "Next scan: open (any consecutive pair)",
      "#a78bfa"
    );
  }

  function fullReset() {
    resetSequence(null);
    totalProfit = 0; recoveryLoss = 0; lastPayoutRatio = null;
    currentStake = 0; lastBalance = null; ladder = 0; stakeLevelIdx = 0;
    tickBuffer.length = 0;
  }

  // ─────────────────────────────────────────────────────────────────────────
  // PROPOSAL / TRADE
  // ─────────────────────────────────────────────────────────────────────────

  function buildProposalVariants(barrier, amount) {
    const base = {
      proposal: 1, contract_type: "DIGITDIFF", currency: "USD",
      amount, basis: "stake", duration: 1, duration_unit: "t", barrier
    };
    return [
      { ...base, underlying_symbol: SYMBOL },
      { ...base, underlying: SYMBOL },
      { ...base, symbol: SYMBOL },
      { ...base }
    ];
  }

  function sendMessage(data) {
    if (!ws || ws.readyState !== WebSocket.OPEN) {
      appendLogLine("WS not open; cannot send.", "red");
      return false;
    }
    ws.send(JSON.stringify(data));
    return true;
  }

  function handleValidationError(errorText) {
    if (!waitingProposal || !proposalVariants) return false;
    const lower = String(errorText || "").toLowerCase();
    if (["underlying_symbol","underlying","properties not allowed","symbol","missing","invalid","validation failed"]
        .some(k => lower.includes(k))) {
      appendLogLine("Proposal validation failed; retrying next variant.", "orange");
      sendNextProposalVariant();
      return true;
    }
    return false;
  }

  function sendNextProposalVariant() {
    if (!proposalVariants) return;
    if (proposalAttempt >= proposalVariants.length) {
      appendLogLine("All proposal variants failed.", "red");
      waitingProposal = false; proposalVariants = null; proposalAttempt = 0;
      return;
    }
    const payload = proposalVariants[proposalAttempt++];
    const label = payload.underlying_symbol ? "underlying_symbol"
      : payload.underlying ? "underlying" : payload.symbol ? "symbol" : "none";
    appendLogLine(`Proposal attempt ${proposalAttempt}: ${label}`, "#a78bfa");
    sendMessage(payload);
  }

  function placeTrade(barrier) {
    if (waitingProposal) { appendLogLine("Already waiting for a proposal.", "orange"); return; }
    pendingBarrier = barrier; probing = true;
    proposalVariants = buildProposalVariants(barrier, baseStakeAmount());
    proposalAttempt = 0; waitingProposal = true;
    appendLogLine(`Probing live payout → DIGITDIFF barrier=${barrier}`, "#38bdf8");
    sendNextProposalVariant();
  }

  // ─────────────────────────────────────────────────────────────────────────
  // TICK ENGINE
  // ─────────────────────────────────────────────────────────────────────────

  function onTick(price) {
    if (!running || paused) return;
    const d = digitFromPrice(price);
    if (d === null) return;

    if (priceEl)     priceEl.textContent     = Number(price).toFixed(2);
    if (lastDigitEl) lastDigitEl.textContent = d;
    tickBuffer.push({ price, digit: d });
    startTickFlush();

    if (captureNextTick) { settlementDigit = d; captureNextTick = false; }
    if (waitingProposal || activeContractId) return;

    if (phase === "scan_ab") {
      if (seqA === null) {
        if (targetPair) { if (d === targetPair[0]) seqA = d; }
        else            { if (d >= 0 && d <= 8)    seqA = d; }
        return;
      }
      if (d === seqA + 1) {
        if (targetPair && (seqA !== targetPair[0] || d !== targetPair[1])) {
          seqA = (d >= 0 && d <= 8) ? d : null; return;
        }
        seqB = d; phase = "scan_x";
        appendLogLine(`Pair [${seqA},${seqB}] found → waiting for x`, "#38bdf8");
      } else {
        seqA = targetPair ? (d === targetPair[0] ? d : null) : (d >= 0 && d <= 8 ? d : null);
      }
      return;
    }

    if (phase === "scan_x") {
      if (d === 9) { appendLogLine("x=9 invalid; restarting.", "red"); resetSequence(targetPair); return; }
      const barrier = d + 1;
      appendLogLine(`Pattern [${seqA},${seqB},${d} → barrier=${barrier}] → BUY`, "#22c55e");
      placeTrade(barrier);
    }
  }

  // ─────────────────────────────────────────────────────────────────────────
  // WEBSOCKET — direct connect + authorize with OAuth token
  // ─────────────────────────────────────────────────────────────────────────

  function connect() {
    if (!activeToken) {
      appendLogLine("Not logged in. Please login first.", "red");
      return;
    }

    resetSequence(null);

    if (ws) {
      ws.onopen = ws.onmessage = ws.onclose = ws.onerror = null;
      ws.close(); ws = null;
    }

    appendLogLine(`Connecting as ${activeLoginid} …`, "yellow");

    ws = new WebSocket(WS_ENDPOINT);

    // Step 1 — open → send token
    ws.onopen = () => {
      appendLogLine("WebSocket open. Authorising …", "#38bdf8");
      ws.send(JSON.stringify({ authorize: activeToken }));
    };

    // Step 2 — handle all server messages
    ws.onmessage = e => {
      let payload;
      try { payload = JSON.parse(e.data); }
      catch { appendLogLine("Bad JSON from server.", "red"); return; }

      if (payload.error) {
        const msg = payload.error.message || JSON.stringify(payload.error);
        appendLogLine(`Error: ${msg}`, "red");
        if (!handleValidationError(msg)) {
          waitingProposal = false; proposalVariants = null; proposalAttempt = 0;
        }
        return;
      }

      switch (payload.msg_type) {

        // ── Auth success ──────────────────────────────────────────────────
        case "authorize": {
          const info = payload.authorize;
          appendLogLine(
            `✓ Authorised — ${info.loginid} | ${info.account_type} | ${Number(info.balance).toFixed(2)} ${info.currency}`,
            "lime"
          );
          updateBalance(Number(info.balance));
          sendMessage({ ticks: SYMBOL, subscribe: 1 });
          sendMessage({ balance: 1, subscribe: 1 });
          break;
        }

        case "tick":
          onTick(payload.tick.quote);
          break;

        case "balance":
          if (payload.balance?.balance !== undefined)
            updateBalance(Number(payload.balance.balance));
          break;

        case "proposal": {
          if (!waitingProposal) break;
          waitingProposal = false; proposalVariants = null; proposalAttempt = 0;

          const prop = payload.proposal;
          if (!prop) { appendLogLine("Empty proposal response.", "red"); break; }

          const probeAsk    = Number(prop.ask_price || 0);
          const probePayout = Number(prop.payout    || 0);

          if (probePayout > 0 && probeAsk > 0) {
            lastPayoutRatio = probePayout / probeAsk;
            appendLogLine(
              `Payout ratio: ${lastPayoutRatio.toFixed(4)} (payout=$${probePayout} / stake=$${probeAsk})`,
              "#38bdf8"
            );
          }

          if (probing) {
            probing = false;
            if (stakeMode === "switch") {
              currentStake = probeAsk;
              const lbl = `Lvl ${stakeLevelIdx+1}/${parsedStakeLevels().length} ($${currentSwitchStake()})`;
              appendLogLine(`TRADE barrier=${pendingBarrier} stake=$${currentStake} [${lbl}]`, "lime");
              sendMessage({ buy: prop.id, price: prop.ask_price });
            } else {
              const correct = recoveryStake(lastPayoutRatio);
              if (Math.abs(correct - probeAsk) < 0.01) {
                currentStake = probeAsk;
                appendLogLine(`TRADE barrier=${pendingBarrier} stake=$${currentStake}`, "lime");
                sendMessage({ buy: prop.id, price: prop.ask_price });
              } else {
                appendLogLine(`Recovery stake=$${correct} (probe=$${probeAsk}) → re-probe`, "orange");
                currentStake = correct;
                proposalVariants = buildProposalVariants(pendingBarrier, correct);
                proposalAttempt = 0; waitingProposal = true;
                sendNextProposalVariant();
              }
            }
          } else {
            currentStake = probeAsk;
            appendLogLine(`TRADE barrier=${pendingBarrier} stake=$${currentStake}`, "lime");
            sendMessage({ buy: prop.id, price: prop.ask_price });
          }
          break;
        }

        case "buy":
          activeContractId = payload.buy?.contract_id || null;
          settlementDigit  = null;
          captureNextTick  = true;
          if (activeContractId)
            sendMessage({ proposal_open_contract: 1, contract_id: activeContractId, subscribe: 1 });
          break;

        case "proposal_open_contract": {
          const c = payload.proposal_open_contract;
          if (!c) return;

          if (profitEl) profitEl.textContent = Number(c.profit || 0).toFixed(2);
          if (typeof c.balance_after === "number" && c.balance_after > 0)
            updateBalance(c.balance_after);

          if (c.is_sold) {
            const pnl = Number(c.profit || 0);
            totalProfit += pnl;
            if (profitEl) profitEl.textContent = totalProfit.toFixed(2);

            const exitDigit   = c.exit_tick !== undefined ? digitFromPrice(c.exit_tick) : null;
            const resultDigit = settlementDigit !== null ? settlementDigit : exitDigit;
            const nextTarget  = resultDigit !== null ? nextPairFromResultDigit(resultDigit) : null;

            if (pnl >= 0) {
              recoveryLoss = 0; currentStake = 0; ladder = 0;
              if (stakeMode === "switch") stakeLevelIdx = 0;
              appendLogLine(
                `WIN +${pnl.toFixed(2)}` +
                (stakeMode === "switch" ? " → Lvl reset" : "") +
                (resultDigit !== null ? ` digit=${resultDigit}` : "") +
                (nextTarget ? ` → next [${nextTarget[0]},${nextTarget[1]}]` : ""),
                "lime"
              );
            } else {
              ladder++;
              if (stakeMode === "switch") {
                const prev = stakeLevelIdx;
                stakeLevelIdx = Math.min(stakeLevelIdx + 1, parsedStakeLevels().length - 1);
                const atMax = stakeLevelIdx === parsedStakeLevels().length - 1;
                appendLogLine(
                  `LOSS $${Math.abs(pnl).toFixed(2)} → Lvl ${prev+1}→${stakeLevelIdx+1} next=$${currentSwitchStake()}` +
                  (atMax ? " [MAX]" : "") +
                  (resultDigit !== null ? ` digit=${resultDigit}` : "") +
                  (nextTarget ? ` → next [${nextTarget[0]},${nextTarget[1]}]` : ""),
                  "red"
                );
              } else {
                recoveryLoss += Math.abs(pnl);
                appendLogLine(
                  `LOSS $${Math.abs(pnl).toFixed(2)} | cumLoss=$${recoveryLoss.toFixed(2)} nextStake≈$${recoveryStake(lastPayoutRatio).toFixed(2)}`,
                  "red"
                );
              }
            }

            if (levelEl) levelEl.textContent = ladder;
            sendMessage({ balance: 1, subscribe: 1 });
            resetSequence(nextTarget);
          }
          break;
        }

        default: break;
      }
    };

    ws.onclose = ev => {
      appendLogLine(`WebSocket closed (code ${ev.code}).`, "orange");
      stopTickFlush();
    };

    ws.onerror = () => {
      appendLogLine("WebSocket error. Check connection.", "red");
    };
  }

  // ─────────────────────────────────────────────────────────────────────────
  // BUTTON HANDLERS
  // ─────────────────────────────────────────────────────────────────────────

  loginBtn.onclick = () => { window.location.href = OAUTH_URL; };

  logoutBtn.onclick = () => {
    running = false;
    if (ws) ws.close();
    stopTickFlush();
    clearSession();
    showLoginScreen();
  };

  startBtn.onclick = () => {
    running = true;
    connect();
    appendLogLine("BOT STARTED", "lime");
  };

  pauseBtn.onclick = () => {
    paused = !paused;
    appendLogLine(paused ? "PAUSED" : "RESUMED", "yellow");
  };

  stopBtn.onclick = () => {
    running = false;
    if (ws) ws.close();
    stopTickFlush();
    appendLogLine("STOPPED", "red");
  };

  resetBtn.onclick = () => {
    fullReset();
    if (profitEl)  profitEl.textContent  = "0.00";
    if (levelEl)   levelEl.textContent   = "0";
    if (balanceEl) balanceEl.textContent = "-";
    appendLogLine("RESET", "orange");
  };

  demoBtn.onclick = () => {
    const demo = accounts.find(a => a.is_virtual);
    if (!demo) { appendLogLine("No demo account found.", "orange"); return; }
    accountSelector.value = accounts.indexOf(demo);
    applySelectedAccount();
    if (ws) { ws.close(); ws = null; }
    appendLogLine("Switched to DEMO — press Start.", "#3b82f6");
  };

  liveBtn.onclick = () => {
    const live = accounts.find(a => !a.is_virtual);
    if (!live) { appendLogLine("No real account found.", "orange"); return; }
    accountSelector.value = accounts.indexOf(live);
    applySelectedAccount();
    if (ws) { ws.close(); ws = null; }
    appendLogLine("Switched to LIVE — press Start.", "#ef4444");
  };

  window.addEventListener("beforeunload", () => { if (ws) ws.close(); stopTickFlush(); });

  // ── Boot ──────────────────────────────────────────────────────────────────
  init();
});
