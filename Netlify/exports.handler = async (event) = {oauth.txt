exports.handler = async (event) => {
  if (event.httpMethod === "OPTIONS") {
    return { statusCode: 200, headers: corsHeaders(), body: "" };
  }

  try {
    const body = JSON.parse(event.body || "{}");
    const { code, redirect_uri } = body;

    if (!code || !redirect_uri) {
      return respond(400, { error: "Missing required parameters: code, redirect_uri" });
    }

    // Deriv token endpoint requires form-encoded body — NOT JSON
    const formBody = new URLSearchParams({
      grant_type:   "authorization_code",
      code,
      client_id:    "33uJq820h9LseRDAZ80iZ",
      redirect_uri
    });

    const response = await fetch("https://oauth.deriv.com/oauth2/token", {
      method:  "POST",
      headers: { "Content-Type": "application/x-www-form-urlencoded" },
      body:    formBody.toString()
    });

    const data = await response.json();

    if (!response.ok || data.error) {
      return respond(400, { error: data.error || "Token exchange failed", detail: data });
    }

    if (!data.access_token) {
      return respond(400, { error: "No access_token returned by Deriv", detail: data });
    }

    return respond(200, { access_token: data.access_token });

  } catch (err) {
    return respond(500, { error: err.message });
  }
};

function corsHeaders() {
  return {
    "Content-Type":                "application/json",
    "Access-Control-Allow-Origin":  "*",
    "Access-Control-Allow-Methods": "POST, OPTIONS",
    "Access-Control-Allow-Headers": "Content-Type"
  };
}

function respond(statusCode, data) {
  return {
    statusCode,
    headers: corsHeaders(),
    body:    JSON.stringify(data)
  };
}
