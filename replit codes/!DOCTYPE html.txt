  <script>
    (function () {
      var params = new URLSearchParams(window.location.search);
      var token1 = params.get("token1");
      var acct1  = params.get("acct1");

      // ── CASE 1: OAuth popup redirect (token in URL) ──────────────
      if (token1) {
        if (window.opener && !window.opener.closed) {
          // Send token back to the bot app that opened this popup
          window.opener.postMessage(
            { type: "deriv_token", token: token1, account: acct1 },
            "*"
          );
          document.getElementById("card").innerHTML =
            '<div class="icon">✅</div>' +
            '<h1 style="color:#10b981">Authorized!</h1>' +
            '<p>Connected to Deriv successfully.<br>This window will close automatically.</p>' +
            '<div class="status"><span class="dot"></span>Sending token to bot…</div>';
          setTimeout(function () { window.close(); }, 1800);
        } else {
          // Direct navigation (not a popup) — show guidance
          window.history.replaceState({}, "", window.location.pathname);
          document.getElementById("card").innerHTML =
            '<div class="icon">⚠️</div>' +
            '<h1 style="color:#f59e0b">Window not found</h1>' +
            '<p>Please go back to the bot app and click<br>' +
            '<strong>Login with Deriv</strong> again to open a proper popup.</p>';
        }
        return;
      }

      // ── CASE 2: Regular direct visit (no token) ──────────────────
      document.getElementById("card").innerHTML =
        '<div class="icon">🤖</div>' +
        '<h1>PRO Deriv Bot</h1>' +
        '<p>This page is the OAuth relay for the PRO Deriv Bot.<br><br>' +
        'Open the bot in your browser and click ' +
        '<strong style="color:#38bdf8">Login with Deriv</strong> to connect.</p>';
    })();
  </script>
